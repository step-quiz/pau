\section*{\textcolor{blauPista}{Qüestió 2}}

\noindent Considereu les matrius
\[
P = \begin{pmatrix} 2 & 1 & -1 \\ 2 & 1 & 0 \\ 1 & 0 & 1 \end{pmatrix}, \quad
Q = \begin{pmatrix} 2 & 2 & 2 \\ 2 & 2 & 2 \\ 2 & 2 & 2 \end{pmatrix}, \quad
R = \begin{pmatrix} 1 & 2 & 3 \\ 1 & 1 & 1 \\ 3 & 2 & 1 \end{pmatrix}.
\]

\subsection*{\textit{a)} \normalfont Decidiu si la matriu $P$ és invertible i, en cas de ser-ho, calculeu la seva inversa. Expliqueu detalladament el procediment seguit. \hspace*{\fill}\textnormal{[1,25~punts]}}

\pista{Una matriu és invertible si i només si el seu determinant és no nul.
\begin{itemize}
\item Calcula $\det(P)$ per decidir si $P$ és invertible.
\item Si ho és, calcula la inversa pel mètode amb què tinguis més pràctica (adjunts o Gauss).
\end{itemize}

\medskip
\textbf{Atenció:} escriu-ne tots els passos. L'enunciat ho demana explícitament i, a la correcció, un resultat sense procediment no puntua.}

\subsection*{\textit{b)} \normalfont Calculeu una matriu $X$ de 3 files i 3 columnes que compleixi $PX + Q = 2R$. \hspace*{\fill}\textnormal{[1,25~punts]}}

\pista{Aïlla $X$ de l'equació matricial:
\begin{itemize}
\item passa $Q$ a l'altre membre;
\item multiplica els dos membres per $P^{-1}$;
\item per als càlculs, aprofita la $P^{-1}$ de l'apartat \textit{a)}.
\end{itemize}

\medskip
\textbf{Atenció:} per l'esquerra o per la dreta? Com que el producte de matrius no és commutatiu, fixa't a quin costat de $X$ hi ha la $P$.}
