\section*{\textcolor{blauPista}{Exercici 4 --- Opció B}}

\noindent Considereu les matrius
\[
A = \begin{pmatrix} a & 0 \\ 1 & a^{2} \end{pmatrix}, \text{ on } a \text{ és un paràmetre real, i } B = \begin{pmatrix} 2 & 0 \\ 1 & 4 \end{pmatrix}.
\]

\subsection*{\textit{a)} \normalfont Trobeu els valors de $a$ que fan que les matrius $A$ i $B$ commutin. (Dues matrius $A$ i $B$ commuten si $AB = BA$.) \hfill\textnormal{[0,75~punts]}}

\pista{Calcula els dos productes $AB$ i $BA$ i iguala'ls element per element.
\begin{itemize}
\item De les quatre igualtats, algunes seran trivialment certes.
\item Les altres et donaran una equació d'on deduiràs el valor o els valors d'$a$.
\end{itemize}

\medskip
\textbf{Atenció:} en el producte de matrius l'ordre dels factors importa; en general, $AB \neq BA$.}

\subsection*{\textit{b)} \normalfont Trobeu els valors de $a$ per als quals la matriu $A$ és invertible. \hfill\textnormal{[0,75~punts]}}

\pista{Calcula $\det(A)$ en funció d'$a$.

\medskip
Recorda: una matriu és invertible si i només si el seu determinant és no nul.}

\subsection*{\textit{c)} \normalfont Trobeu els valors de $a$ per als quals $A^{-1} = A$. \hfill\textnormal{[1~punt]}}

\pista{Evita la inversa: multiplica els dos membres de $A^{-1} = A$ per la matriu $A$.
\begin{itemize}
\item Obtindràs una equació sense la inversa, més fàcil de treballar.
\item Iguala les dues matrius element per element.
\end{itemize}

\medskip
\textbf{Atenció:} t'hi sortiran diverses equacions, i el paràmetre $a$ les ha de complir \emph{totes} alhora.}
